    @classmethod
    def load(cls, path: str, env: Optional[GymEnv] = None, device: Union[torch.device, str] = "auto", custom_objects: Optional[Dict[str, Any]] = None, **kwargs):
        """
        Growth-Aware Load: Automatically handles structural expansion (more layers/heads).
        Supports transitioning from Non-Recurrent to Recurrent and extending feature/action spaces.
        """
        if custom_objects is None:
            custom_objects = {}

        from src.agent.models.universal_recurrent_policy import UniversalRecurrentPolicy
        
        # Determine target policy class
        if "policy_class" not in custom_objects:
             custom_objects["policy_class"] = UniversalRecurrentPolicy
        
        if "policy_kwargs" not in custom_objects:
            custom_objects["policy_kwargs"] = {}
        if "lstm_hidden_size" not in custom_objects["policy_kwargs"]:
            # Default to 512 to match common features_dim
            custom_objects["policy_kwargs"]["lstm_hidden_size"] = 512

        # Ensure optimizer is not loaded during structural transitions
        custom_objects["optimizer"] = None

        try:
            # Attempt standard SB3 load (works if architecture is identical)
            return super().load(path, env=env, device=device, custom_objects=custom_objects, **kwargs)
        except Exception as e:
            logger.warning(f"Standard load failed, attempting growth-aware reconstruction: {e}")
            
            # RECOVERY PATH: Manually reconstruct the model
            from stable_baselines3.common.save_util import load_from_zip_file
            data, params, pytorch_variables = load_from_zip_file(path, device=device, custom_objects=custom_objects)
            
            # Update data with current requirements
            data["policy_class"] = custom_objects["policy_class"]
            if "policy_kwargs" not in data:
                data["policy_kwargs"] = {}
            if "policy_kwargs" in custom_objects:
                data["policy_kwargs"].update(custom_objects["policy_kwargs"])

            # 1. Create model instance
            model = cls(
                policy=data["policy_class"],
                env=env,
                device=device,
                _init_setup_model=False,
                **kwargs
            )

            # Force features_extractor_class for Universal transitions if missing
            if isinstance(model.observation_space, gym.spaces.Dict):
                from src.agent.models.unit_control_network import UnitControlNetwork
                if data["policy_kwargs"].get("features_extractor_class") is None:
                    data["policy_kwargs"]["features_extractor_class"] = UnitControlNetwork

            # 2. Rebuild architecture
            model.__dict__.update(data)
            model._setup_model()
            
            # 3. Growth-Aware Weight Remapping
            policy_params = params.get("policy", {})
            current_policy_sd = model.policy.state_dict()
            new_policy_sd = {}
            
            # Identify topology (Shared vs Separate extractors)
            has_shared_in_model = "features_extractor.cnn.0.weight" in current_policy_sd
            
            # Remap and check every parameter
            for k, v in policy_params.items():
                target_keys = [k]
                
                # Handling shared/split extractor transitions
                if k.startswith("pi_features_extractor.") and has_shared_in_model:
                    target_keys = [k.replace("pi_features_extractor.", "features_extractor.")]
                elif k.startswith("features_extractor.") and not has_shared_in_model:
                    target_keys = [
                        k.replace("features_extractor.", "pi_features_extractor."),
                        k.replace("features_extractor.", "vf_features_extractor.")
                    ]

                for target_key in target_keys:
                    if target_key not in current_policy_sd:
                        continue
                        
                    curr_v = current_policy_sd[target_key]
                    
                    # CASE 1: Exact Match
                    if v.shape == curr_v.shape:
                        new_policy_sd[target_key] = v
                        
                    # CASE 2: CNN Channel Growth (Input Channels changed)
                    elif len(v.shape) == 4 and v.shape[1] < curr_v.shape[1] and \
                         v.shape[0] == curr_v.shape[0] and v.shape[2:] == curr_v.shape[2:]:
                        logger.info(f"Expanding CNN input channels for {target_key}: {v.shape[1]} -> {curr_v.shape[1]}")
                        # Create new weight tensor and copy old channels
                        expanded_v = curr_v.clone() # Keep random/zero init for new channels
                        expanded_v[:, :v.shape[1], :, :] = v
                        new_policy_sd[target_key] = expanded_v
                        
                    # CASE 3: Action Head Growth (Output Logits changed)
                    elif "action_net" in target_key and len(v.shape) == 2 and v.shape[0] != curr_v.shape[0]:
                        # MultiDiscrete logits are concatenated [N_Heads + 1024]
                        # We try to load what we can if the spatial part (usually 1024) is trailing
                        logger.warning(f"Attempting partial load for action head {target_key} ({v.shape[0]} -> {curr_v.shape[0]})")
                        expanded_v = curr_v.clone()
                        min_logits = min(v.shape[0], curr_v.shape[0])
                        expanded_v[:min_logits, :] = v[:min_logits, :]
                        new_policy_sd[target_key] = expanded_v
                    
                    else:
                        logger.debug(f"Skipping mismatched param {target_key}: {v.shape} vs {curr_v.shape}")

            # Load the partial state dict
            model.policy.load_state_dict(new_policy_sd, strict=False)
            
            # 4. Fresh Optimizer
            model._setup_lr_schedule()
            model.policy.optimizer = model.policy.optimizer_class(
                model.policy.parameters(), lr=model.learning_rate, **model.policy.optimizer_kwargs
            )
            
            logger.info("Growth-aware reconstruction complete.")
            return model
